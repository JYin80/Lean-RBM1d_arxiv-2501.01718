\chapter{The model}

\begin{definition}[Block covariance matrix, \S2.1]\label{def:SB}
  \lean{RBM.SB}\leanok
  Let $L\ge 3$ and work on $\ZL$.  Set
  \[ \SB_{ab} \;=\; \tfrac13\,\mathbf 1\!\left(\operatorname{dist}_{\ZL}(a,b)\le 1\right),
     \qquad a,b\in\ZL . \]
  Equivalently $\SB$ is the circulant matrix generated by the kernel supported on
  $\{0,1,-1\}$ with value $1/3$.
\end{definition}

\begin{remark}\label{rem:L3}
  The paper leaves the hypothesis $L\ge 3$ implicit.  It is needed: for $L=1$ the
  row sums are $1/3$, and for $L=2$ one has $1=-1$ in $\ZL$ so the row sums are
  $2/3$.  See \texttt{docs/paper-deltas.md}.
\end{remark}

\begin{lemma}[Elementary properties of $\SB$]\label{lem:SB-basic}
  \lean{RBM.SB_isSymm, RBM.SB_apply_add_right, RBM.sum_SB_row, RBM.SB_mulVec_one}
  \leanok\uses{def:SB}
  $\SB$ is symmetric and translation invariant, each row sums to $1$, and
  $\SB \mathbf 1 = \mathbf 1$.
\end{lemma}
\begin{proof}\leanok\uses{def:SB}
  Symmetry and translation invariance are immediate from the circulant form.  The
  row sums reduce, after the change of variables $b\mapsto a-b$, to the total mass
  of the kernel, which is $3\cdot\frac13=1$ because $0,1,-1$ are distinct in $\ZL$
  when $L\ge 3$.
\end{proof}

\begin{definition}[The band model, \S2.1 and (2.5)]\label{def:model}
  \lean{RBM.SW, RBM.Svar, RBM.Eblk, RBM.Iblk, RBM.Spaper, RBM.Epaper, RBM.splitEquiv}
  \leanok\uses{def:SB}
  With $N=WL$ and blocks $I_a=\{aW,\dots,aW+W-1\}\subseteq\mathbb Z_N$, $a\in\ZL$,
  \[ S_{ij}=\frac1{3W}\sum_a \mathbf 1(i\in I_a)\,\mathbf 1(j\in I_a\cup I_{a+1}\cup I_{a-1}),
     \qquad (E_a)_{ij}=\delta_{ij}W^{-1}\mathbf 1(i\in I_a). \]
\end{definition}

\begin{lemma}[$S=\SB\otimes S_W$ and properties of $E_a$]\label{lem:model}
  \lean{RBM.Spaper_eq, RBM.Epaper_eq, RBM.split_bijective, RBM.sum_Svar_row,
    RBM.Svar_transpose, RBM.sum_Eblk, RBM.Eblk_conjTranspose, RBM.Eblk_mul_Eblk}
  \leanok\uses{def:model, lem:SB-basic}
  Under $i\mapsto(\lfloor i/W\rfloor, i\bmod W)$, $S=\SB\otimes S_W$ with
  $(S_W)_{\alpha\beta}=W^{-1}$.  $S$ is symmetric with rows summing to $1$;
  $E_a$ is self-adjoint, $E_aE_b=\delta_{ab}W^{-1}E_a$, and $\sum_a E_a=W^{-1}I$.
\end{lemma}
\begin{proof}\leanok\uses{def:model, lem:SB-basic}
  $i\in I_a$ iff $\lfloor i/W\rfloor=a$, so the sum over $a$ collapses to
  $a=\lfloor i/W\rfloor$; the remaining indicator is $\SB_{\lfloor i/W\rfloor\lfloor j/W\rfloor}$.
\end{proof}

\begin{lemma}[Semicircle law and the $z_t$ flow, Definition 2.7 and Lemma 2.8]\label{lem:2.8}
  \lean{RBM.mE, RBM.mE_mul, RBM.norm_mE, RBM.eq_mE, RBM.msc, RBM.msc_mul, RBM.msc_im_pos,
    RBM.norm_msc_lt_one, RBM.zt, RBM.zt_im, RBM.lemE, RBM.lemT, RBM.mE_lemE, RBM.lemT_eq,
    RBM.lemT_pos, RBM.lemT_lt_one, RBM.abs_lemE_lt_two, RBM.msc_eq_sqrt_mul_mE,
    RBM.eq_inv_sqrt_mul_zt}
  \leanok
  $m^{(E)}=(-E+i\sqrt{4-E^2})/2$ is the unique root of $m(m+E)=-1$ with $\operatorname{Im}m>0$,
  and $|m^{(E)}|=1$.  For $\operatorname{Im}z>0$, $|m_{sc}(z)|<1$; with
  $E=-2\operatorname{Re}m_{sc}/|m_{sc}|$ and $t=|m_{sc}|^2$ one has
  $m^{(E)}=m_{sc}/|m_{sc}|$, $t=m_{sc}^2/(m^{(E)})^2\in(0,1)$, $|E|<2$,
  $m_{sc}(z)=t^{1/2}m^{(E)}$ (2.38) and $z=t^{-1/2}z^{(E)}_t$ (2.37), where
  $z^{(E)}_t=E+(1-t)m^{(E)}$.
\end{lemma}
\begin{proof}\leanok
  The roots of $m^2+zm+1$ multiply to $1$, so their imaginary parts have opposite signs,
  and they sum to $-z$.  $u=m_{sc}/|m_{sc}|$ satisfies $u(u+E)=-|u|^2=-1$; uniqueness gives
  $u=m^{(E)}$, and (2.37) follows from $E+u=-1/u$.  The $\kappa$-dependent bounds (2.40)
  are deferred to the stochastic layer.
\end{proof}

\begin{definition}[Deterministic domination, Definition 2.1 (ii)]\label{def:2.1-ii}
  \lean{RBM.UnifDetDom, RBM.DetDom, RBM.detDom_iff}\leanok
  For deterministic $N$-dependent quantities $f,g$ write $f\prec g$ if
  $f\le N^\tau g$ for every $\tau>0$ and all sufficiently large $N$.  The Lean
  definition is uniform in a possibly $N$-dependent parameter $u\in U(N)$, as in
  part (i) of the definition.
\end{definition}

\begin{lemma}[Closure properties of $\prec$]\label{lem:2.1-ii-closure}
  \lean{RBM.UnifDetDom.refl, RBM.UnifDetDom.trans, RBM.UnifDetDom.add,
    RBM.UnifDetDom.mul, RBM.UnifDetDom.const_mul_left, RBM.UnifDetDom.const_mul_right,
    RBM.UnifDetDom.of_eventually_le_const_mul}
  \leanok\uses{def:2.1-ii}
  On non-negative quantities $\prec$ is reflexive and transitive, closed under sums
  and products, and absorbs positive constant factors on either side; $f\le Cg$
  implies $f\prec g$.
\end{lemma}
\begin{proof}\leanok\uses{def:2.1-ii}
  Split $N^\tau = N^{\tau/2}N^{\tau/2}$ for transitivity and products; for
  constants use $N^\tau\to\infty$.
\end{proof}

\chapter{The propagator}

\begin{definition}[Propagator, Definition 2.13]\label{def:Theta}
  \lean{RBM.Theta}\leanok\uses{def:SB}
  For $\xi\in\CC$ with $|\xi|<1$ set $\Th_\xi := (1-\xi\SB)^{-1}$.
\end{definition}

\begin{lemma}[Operator norm of $\SB$]\label{lem:norm-SB}
  \lean{RBM.norm_SB}\leanok\uses{lem:SB-basic}
  $\|\SB\|_{\ell^\infty\to\ell^\infty}=1$.
\end{lemma}
\begin{proof}\leanok\uses{lem:SB-basic}
  The $\ell^\infty$ operator norm is the maximum $\ell^1$ row norm; all entries are
  non-negative and every row sums to $1$.
\end{proof}

\begin{lemma}[$\Th_\xi$ is a two-sided inverse]\label{lem:Theta-inv}
  \lean{RBM.Theta_mul, RBM.mul_Theta, RBM.eq_Theta_of_mul}\leanok
  \uses{def:Theta, lem:norm-SB}
  For $|\xi|<1$ the matrix $1-\xi\SB$ is invertible and $\Th_\xi$ is its two-sided
  inverse.  Moreover any left inverse of $1-\xi\SB$ equals $\Th_\xi$.
\end{lemma}
\begin{proof}\leanok\uses{lem:norm-SB}
  $\|\xi\SB\|=|\xi|<1$, so the geometric series converges and $1-\xi\SB$ is a unit.
\end{proof}

\begin{lemma}[Lemma 2.14 (1): symmetry]\label{lem:Theta-sym}
  \lean{RBM.Theta_transpose}\leanok\uses{lem:Theta-inv, lem:SB-basic}
  $(\Th_\xi)_{xy}=(\Th_\xi)_{yx}$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  $\Th^{\mathsf T}(1-\xi\SB) = \Th^{\mathsf T}(1-\xi\SB)^{\mathsf T}
   = \bigl((1-\xi\SB)\Th\bigr)^{\mathsf T} = 1$, so $\Th^{\mathsf T}$ is a left
  inverse; conclude by uniqueness.
\end{proof}

\begin{lemma}[Lemma 2.14 (2): translation invariance]\label{lem:Theta-transl}
  \lean{RBM.Theta_apply_add_right}\leanok\uses{lem:Theta-inv, lem:SB-basic}
  $(\Th_\xi)_{xy}=(\Th_\xi)_{x+1,y+1}$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Conjugating by the shift, i.e.\ taking the submatrix along $a\mapsto a+c$,
  fixes $1-\xi\SB$; conclude by uniqueness of the inverse.
\end{proof}

\begin{lemma}[Lemma 2.14 (3): commutativity]\label{lem:Theta-comm}
  \lean{RBM.Theta_commute_SB, RBM.Theta_commute}\leanok\uses{lem:Theta-inv}
  $[\SB,\Th_\xi]=[\Th_{\xi'},\Th_\xi]=0$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Inverses of commuting elements commute.
\end{proof}

\begin{lemma}[Lemma 2.14 (5): random walk representation]\label{lem:Theta-series}
  \lean{RBM.Theta_eq_tsum}\leanok\uses{lem:Theta-inv}
  $\Th_\xi=\sum_{k\ge 0}\xi^k(\SB)^k$, convergent in operator norm.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Neumann series.
\end{proof}

\begin{lemma}[Row sums]\label{lem:Theta-rowsum}
  \lean{RBM.sum_Theta_row, RBM.Theta_mulVec_one}\leanok
  \uses{lem:Theta-inv, lem:SB-basic}
  $\sum_b (\Th_\xi)_{ab} = (1-\xi)^{-1}$.  Used in (3.47)--(3.48) and in Step 4 of
  the proof of Lemma 3.6.
\end{lemma}
\begin{proof}\leanok\uses{lem:SB-basic}
  Apply $\Th_\xi$ to the constant vector and use $\SB\mathbf 1=\mathbf 1$.
\end{proof}

\begin{lemma}[Equation (2.51)]\label{lem:Theta-deriv}
  \lean{RBM.hasDerivAt_Theta_apply, RBM.Theta_sub_Theta}\leanok
  \uses{lem:Theta-inv}
  $\partial_\xi \Th_\xi = \Th_\xi \SB \Th_\xi$, entrywise.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  The resolvent identity
  $\Th_\zeta-\Th_\xi=(\zeta-\xi)\,\Th_\zeta\SB\Th_\xi$
  together with continuity of $\xi\mapsto\Th_\xi$.
\end{proof}

\begin{lemma}[Uniform $\ell^1$ and $\ell^\infty$ bound]\label{lem:Theta-bounds}
  \lean{RBM.norm_Theta_le, RBM.sum_norm_Theta_row_le, RBM.norm_Theta_apply_le,
    RBM.sum_norm_Theta_row_of_real}\leanok
  \uses{lem:Theta-series, lem:norm-SB}
  $\max_a\sum_b |(\Th_\xi)_{ab}| \le (1-|\xi|)^{-1}$, and in particular
  $|(\Th_\xi)_{ab}| \le (1-|\xi|)^{-1}$.  For real $\xi = t$ the entries are positive and
  the $\ell^1$ norm is \emph{exactly} $(1-t)^{-1}$.
  \emph{This is not yet (3.35)--(3.36)}: see Lemma~\ref{lem:3.35}.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-series}
  Sum the Neumann series in operator norm.
\end{proof}

\begin{lemma}[Band structure and a first decay bound]\label{lem:Theta-band}
  \lean{RBM.SB_pow_apply_eq_zero, RBM.Theta_eq_geom_add, RBM.norm_Theta_apply_le_pow}\leanok
  \uses{lem:Theta-series, lem:Theta-bounds}
  $(\SB)^k$ vanishes outside the band of width $k$, and consequently
  \[ \bigl|(\Th_\xi)_{xy}\bigr| \;\le\;
     \frac{|\xi|^{\operatorname{dist}(x,y)}}{1-|\xi|} . \]
  This is exponential decay with length $1/\log(1/|\xi|)$, weaker than (2.52).
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-series}
  Truncate the Neumann series at order $d=\operatorname{dist}(x,y)$:
  $\Th_\xi=\sum_{k<d}(\xi\SB)^k+(\xi\SB)^d\Th_\xi$, and the finite sum has
  vanishing $(x,y)$ entry.
\end{proof}

\begin{lemma}[Fourier representation, Appendix B (B.1)]\label{eq:B.1}
  \lean{RBM.Shat, RBM.Shat_eq_cos, RBM.SB_mulVec_char, RBM.one_sub_mul_Shat_ne_zero,
    RBM.fourierKernel, RBM.Theta_eq_circulant_fourierKernel, RBM.Theta_apply_fourier}
  \leanok\uses{def:Theta, lem:Theta-inv}
  The plane wave $e_p(x)=e^{ipx}$, $p\in\frac{2\pi}{L}\ZL$, is an eigenvector of
  $\SB$ with eigenvalue $\widehat S(p)=(1+2\cos p)/3$.  Since $|\widehat S(p)|\le 1$,
  $1-\xi\widehat S(p)\neq 0$ for $|\xi|<1$, and
  \[ (\Th_\xi)_{xy} \;=\; \frac1L\sum_{p\in\frac{2\pi}{L}\ZL}
     \frac{e^{ip(x-y)}}{1-\xi\widehat S(p)} . \]
\end{lemma}
\begin{proof}\leanok\uses{def:Theta, lem:Theta-inv}
  Call the right-hand side $K(x-y)$.  By the eigenvector relation and orthogonality
  of characters, $K-\xi\,\SB K=\delta_0$; the circulant matrix generated by $K$ is
  therefore a left inverse of $1-\xi\SB$, and uniqueness of the inverse closes it.
\end{proof}

\begin{lemma}[Two-sided bound on the symbol, Appendix B (B.3)]\label{eq:B.3}
  \lean{RBM.theta, RBM.abs_theta_le_pi, RBM.Shat_eq_cos_theta, RBM.le_one_sub_cos,
    RBM.norm_one_sub_mul_Shat_le, RBM.le_norm_one_sub_mul_Shat,
    RBM.norm_one_sub_mul_Shat_asymp}\leanok
  \uses{eq:B.1}
  With $\theta(p)=2\pi\,\tilde p/L\in[-\pi,\pi]$, $\tilde p$ the representative of $p$ closest
  to $0$: uniformly in $L$, $|\xi|<1$ and $p$,
  \[ \frac{|1-\xi|+\theta(p)^2}{6\pi^2} \;\le\; |1-\xi\widehat S(p)| \;\le\;
     |1-\xi|+\theta(p)^2 . \]
\end{lemma}
\begin{proof}\leanok\uses{eq:B.1}
  $s=1-\widehat S=\frac23(1-\cos\theta)$ and $\frac{2}{\pi^2}\theta^2\le1-\cos\theta\le
  \frac12\theta^2$ (Jordan at $\theta/2$).  Upper bound: $1-\xi\widehat S=(1-\xi)+\xi s$.
  Lower bound: if $\widehat S\ge\frac12$, $1-\xi\widehat S=s+\widehat S(1-\xi)$ with
  $\operatorname{Re}(1-\xi)>0$ gives $|1-\xi\widehat S|\ge s$ and
  $\ge\widehat S|1-\xi|-s$; if $\widehat S<\frac12$, $|1-\xi\widehat S|>\frac12$.
\end{proof}

\begin{definition}[Infinite-volume kernel, (B.2)]\label{eq:B.2}
  \lean{RBM.Dxi, RBM.kernelFun, RBM.Kinf}\leanok
  $K_{\xi,\infty}(u)=\frac1{2\pi}\int_{-\pi}^{\pi}\frac{e^{ipu}}{1-\xi\widehat S(p)}\,dp$,
  $u\in\mathbb Z$, with $D_\xi(z)=1-\xi(1+2\cos z)/3$ for complex $z$.
\end{definition}

\begin{lemma}[Shifted symbol, (B.4)]\label{eq:B.4}
  \lean{RBM.norm_cos_add_mul_I_sub_le, RBM.cZero, RBM.le_norm_Dxi, RBM.Dxi_ne_zero}\leanok
  \uses{eq:B.3, eq:B.2}
  With $\kappa=|1-\xi|^{1/2}$ and $c_0=1/(16\pi^2)$: for $|p|\le\pi$ and $|\eta|\le c_0\kappa$,
  $|D_\xi(p+i\eta)|\ge(\kappa^2+p^2)/(12\pi^2)$.
\end{lemma}
\begin{proof}\leanok\uses{eq:B.3, eq:B.2}
  $|\cos(p+i\eta)-\cos p|\le(\cosh\eta-1)+|\sin p||\sinh\eta|\le\eta^2+2|p||\eta|
  \le2c_0(\kappa^2+p^2)$, absorbed by the lower bound $(\kappa^2+p^2)/(6\pi^2)$ of (B.3).
\end{proof}

\begin{lemma}[Contour shift and decay of $K_{\xi,\infty}$, (B.5)]\label{eq:B.5}
  \lean{RBM.kernelFun_add_two_pi, RBM.integral_kernelFun_shift, RBM.norm_kernelFun_shift_le,
    RBM.integral_inv_sq_add_sq_le, RBM.norm_Kinf_le, RBM.norm_Kinf_le_exists}\leanok
  \uses{eq:B.4}
  $|K_{\xi,\infty}(u)|\le\frac{6\pi^2}{\kappa}\,e^{-c_0\kappa|u|}$.
\end{lemma}
\begin{proof}\leanok\uses{eq:B.4}
  The integrand is $2\pi$-periodic and holomorphic on $[-\pi,\pi]\times[0,\eta]$ by (B.4),
  so Cauchy's theorem on the rectangle (vertical sides cancel) moves the contour to
  $\mathbb R+i\eta$, $\eta=c_0\kappa\operatorname{sgn}u$.  There $|e^{i(p+i\eta)u}|=e^{-c_0\kappa|u|}$
  and $\int_{-\pi}^{\pi}dp/(\kappa^2+p^2)\le\pi/\kappa$.
\end{proof}

\begin{lemma}[Periodization, after (B.2)]\label{eq:B.2-per}
  \lean{RBM.integral_exp_int, RBM.Kinf_sub_SB, RBM.summable_Kinf_shift, RBM.perSum, RBM.perK,
    RBM.perK_sub_SB_mulVec, RBM.Theta_eq_circulant_perK, RBM.Theta_apply_periodize}\leanok
  \uses{eq:B.5, def:Theta, lem:Theta-inv}
  $(\Th_\xi)_{xy}=\sum_{n\in\mathbb Z}K_{\xi,\infty}(u+nL)$, $u=x-y$, the sum converging
  absolutely.
\end{lemma}
\begin{proof}\leanok\uses{eq:B.5, def:Theta, lem:Theta-inv}
  Instead of Fourier inversion: $\int_{-\pi}^{\pi}e^{ipw}dp=2\pi\delta_{w,0}$ gives
  $K_\infty-\xi\SB K_\infty=\delta_0$ on $\mathbb Z$; by (B.5) the periodization converges
  absolutely and solves the same equation on $\ZL$, so its circulant inverts $1-\xi\SB$.
\end{proof}

\begin{lemma}[(3.35)--(3.36): short and long edges]\label{lem:3.35}
  \uses{lem:Theta-bounds, lem:Theta-decay, lem:2.8}
  The paper states, with $\ell_t$ the decay length and $\eta_t = \operatorname{Im} z_t$,
  \[ \|\Th_{tm^2}\|_{\max} \prec 1, \quad \|\Th_{t|m|^2}\|_{\max} \prec \frac1{\ell_t\eta_t};
     \qquad
     \|\Th_{tm^2}\|_{1} \prec 1, \quad \|\Th_{t|m|^2}\|_{1} \prec \frac1{\eta_t} . \]
  \emph{Not yet formalized as stated}, and it is worth being precise about what is missing.
  \begin{itemize}
    \item \textbf{Long edge} ($\xi = t|m|^2 = t$, real).  The $\max$ bound follows from
      (2.52) (Lemma~\ref{lem:Theta-decay}) and the $\ell^1$ bound is the identity
      $(1-t)^{-1}$ of Lemma~\ref{lem:Theta-bounds} --- \emph{provided} the dictionary
      $1-t \asymp \eta_t$, which is part of the quantitative half (2.40) of Lemma 2.8
      that is currently deferred.
    \item \textbf{Short edge} ($\xi = tm^2$ with $m$ non-real, $|m| = 1$).  Here
      $|\xi| = t \to 1$, so the uniform bound $(1-|\xi|)^{-1}$ of
      Lemma~\ref{lem:Theta-bounds} \textbf{blows up} and does not give $\prec 1$.  What is
      needed is a lower bound on $|1-tm^2|$, which is where the complex-$\xi$ estimate
      (Lemma~\ref{lem:Theta-decay-complex}) enters.
  \end{itemize}
\end{lemma}

\begin{lemma}[The characteristic root $\rho(\xi)$]\label{lem:rho}
  \lean{RBM.rho, RBM.norm_root_ne_one, RBM.norm_rho_lt_one}\leanok\uses{def:Theta}
  Write $c = 3/\xi - 1$.  The characteristic equation of the three-term recursion
  behind $(1-\xi\SB)\Th_\xi = 1$ is $\rho^2 - c\rho + 1 = 0$, whose two roots
  multiply to $1$.  For $0 < |\xi| < 1$ neither root lies on the unit circle, so
  exactly one has modulus $< 1$; call it $\rho(\xi)$.
\end{lemma}
\begin{proof}\leanok\uses{def:Theta}
  If $|\rho| = 1$ then $|c| = |\rho + \rho^{-1}| \le |\rho| + |\rho^{-1}| = 2$,
  hence $3/|\xi| = |c+1| \le 3$, i.e.\ $|\xi| \ge 1$.  This is the same statement
  as $1 - \xi\widehat S(p) \ne 0$ in the Fourier picture of Appendix B, and it is
  exactly what fails for $|\xi| \ge 1$.
\end{proof}

\begin{lemma}[Closed form of the propagator]\label{lem:closed-form}
  \lean{RBM.theta_eq_circulant, RBM.theta_apply_closed_form}\leanok
  \uses{lem:rho, lem:Theta-inv}
  For $0 < |\xi| < 1$,
  \[ (\Th_\xi)_{xy} = A(\xi)\bigl(\rho^{\,d} + \rho^{\,L-d}\bigr),\qquad
     d = \overline{(x-y)} \in [0,L),\qquad
     A(\xi) = \frac{3\rho}{\xi\,(\rho^L-1)(\rho^2-1)} . \]
\end{lemma}
\begin{proof}\leanok\uses{lem:rho, lem:Theta-inv}
  The kernel satisfies $k(u-1) + k(u+1) - c\,k(u) = -(3/\xi)\delta_{u,0}$.  The
  symmetric shape makes the value at $n=L$ agree with the value at $n=0$, so the
  wrap-around at $d = L-1$ falls under the generic case and $d = 0$ is the only
  special one; the defect there is what fixes $A(\xi)$.  Conclude by uniqueness
  of the inverse.
\end{proof}

\begin{lemma}[Decay rate for real spectral parameter]\label{lem:rho-rate}
  \lean{RBM.rho_real_bounds}\leanok\uses{lem:rho}
  For $0 < t < 1$, $\rho(t)$ is real, lies in $(0,1)$, and
  \[ \sqrt{1-t} \;\le\; 1 - \rho(t) \;\le\; \sqrt3\,\sqrt{1-t} . \]
  So the decay length $1/(1-\rho)$ is of order $(1-t)^{-1/2}$.  This is the
  long-edge case $\xi = t|m|^2$, where $|m| = 1$ makes $\xi$ real; it is the case
  that needs the sharp length.
\end{lemma}
\begin{proof}\leanok\uses{lem:rho}
  The discriminant $(3/t-1)^2 - 4$ is a positive real, so both roots are real and
  $\rho \in (0,1)$; then $1 - \|\rho\| = 1 - \rho$, which is the direction the
  decay estimate needs.  Combine with the exact identity
  $(1-\rho)^2 = (1-\xi)(1+\rho+\rho^2)$ and $1 \le 1+\rho+\rho^2 \le 3$.
\end{proof}

\begin{lemma}[Decay rate for complex spectral parameter]\label{lem:rho-rate-complex}
  \lean{RBM.sub_re_le_of_poly, RBM.three_mul_norm_rho_lt, RBM.sq_one_sub_norm_rho_le,
    RBM.norm_one_sub_xi_le, RBM.rho_complex_bounds}\leanok\uses{lem:rho}
  For all $0<|\xi|<1$,
  \[ \tfrac18\,|1-\xi| \;\le\; (1-|\rho(\xi)|)^2 \;\le\; 3\,|1-\xi| , \]
  so the decay length is $|1-\xi|^{-1/2}$ for every admissible $\xi$, including the
  short-edge case $\xi=tm^2$.
\end{lemma}
\begin{proof}\leanok\uses{lem:rho}
  Upper bound: $1-|\rho|\le|1-\rho|$ and $|1-\rho|^2=|1-\xi|\,|1+\rho+\rho^2|\le3|1-\xi|$.
  Lower bound: $|\xi|<1$ is $3|\rho|<|1+\rho+\rho^2|$.  With $p=\operatorname{Re}\rho$,
  $r=|\rho|$ this reads $g>0$, where exactly
  $g=-4(r-p-(1-r)^2)(r+p)+(r^2-4r+1)(r^2+1-2p)$.  For $r\ge\frac12$ the second term is
  $\le0$, so $r-p\le(1-r)^2$, whence $|1-\rho|^2=(1-r)^2+2(r-p)\le3(1-r)^2$ and
  $|1-\xi|=|1-\rho|^2/|1+\rho+\rho^2|\le2(1-r)^2$.  For $r<\frac12$, $|1-\xi|<2<8(1-r)^2$.
\end{proof}

\begin{lemma}[The prefactor of (2.52)]\label{lem:prefactor}
  \lean{RBM.AA_eq, RBM.pow_mul_one_add_le, RBM.le_one_sub_pow, RBM.ellHat,
    RBM.ellHat_mul_one_sub_le, RBM.norm_AA_le_of_real}
  \leanok\uses{lem:closed-form, lem:rho-rate}
  The constant of the closed form simplifies to
  \[ A(\xi) \;=\; \frac{1-\rho}{(1-\xi)\,(1-\rho^L)\,(1+\rho)} , \]
  and for real $\xi = t \in (0,1)$,
  \[ |A(t)| \;\le\; \frac{4}{(1-t)\,\hat\ell(t)},
     \qquad \hat\ell(t) = \min\bigl((1-t)^{-1/2}, L\bigr). \]
\end{lemma}
\begin{proof}\leanok\uses{lem:closed-form, lem:rho-rate}
  Substituting $(1-\xi)\cdot 3\rho = \xi(1-\rho)^2$ into
  $A = 3\rho/[\xi(\rho^L-1)(\rho^2-1)]$ cancels one factor $1-\rho$ against
  $\rho^2-1$ and converts $1/\xi$ into $1/(1-\xi)$.  For the bound, Bernoulli
  $1+Lx \le (1+x)^L$ together with $(1-x)(1+x)\le 1$ gives
  $1-\rho^L \ge L(1-\rho)/(1+L(1-\rho))$, which is $\asymp \min(1, L(1-\rho))$;
  the two regimes of the $\min$ in $\hat\ell$ are exactly $L(1-\rho)\gtrsim 1$ and
  $L(1-\rho)\lesssim 1$, and $\sqrt{1-t}\le 1-\rho\le\sqrt 3\sqrt{1-t}$ converts
  between them.  No exponentials are needed.
\end{proof}

\begin{lemma}[(2.52) for complex $\xi$, Fourier route]\label{lem:Theta-decay-complex}
  \lean{RBM.tsum_exp_neg_abs_shift_le, RBM.norm_Theta_apply_le_large,
    RBM.norm_Theta_apply_le_small, RBM.norm_Theta_apply_le_complex,
    RBM.norm_Theta_apply_le_exists}\leanok
  \uses{eq:B.2-per, eq:B.5, eq:B.3, eq:B.1}
  For every $|\xi|<1$, with $\hat\ell(\xi)=\min(|1-\xi|^{-1/2},L)$ and $c_0=1/(16\pi^2)$,
  \[ \bigl|(\Th_\xi)_{xy}\bigr|\le\frac{C\,e^{-c_0\|x-y\|/\hat\ell(\xi)}}{|1-\xi|\,\hat\ell(\xi)} ,\]
  $C$ independent of $L$ and $\xi$.
\end{lemma}
\begin{proof}\leanok\uses{eq:B.2-per, eq:B.5, eq:B.3, eq:B.1}
  $\kappa L\ge1$: periodization and (B.5); the terms $n\ge0$ and $n<0$ are geometric series
  from $e^{-c_0\kappa\|u\|}$ with ratio $\le e^{-c_0}$.  $\kappa L<1$: (B.1) and (B.3); the zero
  mode gives $1/(\kappa^2L)$, the rest $\frac32 ZL$ with $Z=\sum_{n\neq0}n^{-2}$, and
  $L\le1/(\kappa^2L)$.
\end{proof}

\begin{lemma}[Lemma 2.14 (4): exponential decay, (2.52)]\label{lem:Theta-decay}
  \lean{RBM.zdist_le, RBM.norm_Theta_apply_le_of_real}
  \leanok\uses{lem:Theta-band, lem:closed-form, lem:rho-rate, lem:prefactor}
  With $\hat\ell(\xi):=\min(|1-\xi|^{-1/2},L)$ and real $\xi = t \in (0,1)$,
  \[ \bigl|(\Th_t)_{xy}\bigr| \;\le\;
     \frac{8e\,\cdot\,e^{-\|x-y\|/\hat\ell(t)}}{(1-t)\cdot\hat\ell(t)} . \]
  The paper's $C$ and $c$ are here the explicit $8e$ and $1$.  Real $\xi$ is the
  long-edge case $\xi = t|m|^2 = t$ of the paper (recall $|m| = 1$), which is the
  case that needs the sharp length scale.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-band, lem:closed-form, lem:rho-rate, lem:prefactor}
  Two regimes.  If $(1-t)^{-1/2}\le L$ the exponential does the work:
  $\rho \le e^{-(1-\rho)} \le e^{-\sqrt{1-t}} = e^{-1/\hat\ell}$.  If $L \le (1-t)^{-1/2}$
  the exponential is useless ($e^{-d/L}\ge e^{-1}$ for $d \le L$) and the prefactor
  bound $|A| \le 4/((1-t)L)$ of Lemma~\ref{lem:prefactor} carries the estimate alone.
  No contour shift and no Poisson summation.
\end{proof}

\begin{lemma}[$\ell^1$ bound on a row, after (2.52)]\label{lem:Theta-l1}
  \lean{RBM.exists_pos_Theta_apply_of_real, RBM.sum_norm_Theta_row_of_real}
  \leanok\uses{lem:closed-form, lem:Theta-rowsum}
  For real $\xi = t \in (0,1)$ every entry of $\Th_t$ is a positive real, hence
  \[ \sum_b \bigl|(\Th_t)_{ab}\bigr| \;=\; \sum_b (\Th_t)_{ab} \;=\; (1-t)^{-1} . \]
  The paper states $\le C(1-t)^{-1}$; positivity turns the inequality into an identity
  with constant $1$.
\end{lemma}
\begin{proof}\leanok\uses{lem:closed-form, lem:Theta-rowsum}
  In the closed form $A(\rho^d+\rho^{L-d})$ the root $\rho$ lies in $(0,1)$ and all four
  factors of $A = (1-\rho)/[(1-t)(1-\rho^L)(1+\rho)]$ are positive, so the $\ell^1$ norm of
  a row equals its signed sum, which is $(1-\xi)^{-1}$ by Lemma~\ref{lem:Theta-rowsum}.
\end{proof}

\begin{lemma}[Difference estimates (2.53)--(2.54)]\label{lem:Theta-diff}
  \lean{RBM.kern_succ_sub, RBM.kern_second_diff, RBM.norm_AA_mul_one_sub_rho_le,
    RBM.norm_AA_mul_one_sub_rho_sq_le, RBM.norm_Theta_sub_shift_le,
    RBM.kern_wrap_second, RBM.norm_Theta_second_diff_le, RBM.norm_Theta_second_diff_diag_le,
    RBM.norm_Theta_second_diff_le_pow, RBM.mul_exp_neg_le_exp_neg_one,
    RBM.norm_Theta_second_diff_le_inv_dist}
  \leanok\uses{lem:Theta-decay, lem:prefactor}
  For real $\xi = t \in (0,1)$,
  \[ \bigl|(\Th_t)_{x,y}-(\Th_t)_{x,y+1}\bigr| \le \frac{8\sqrt3}{\hat\ell(t)\,(1-t)^{1/2}},
     \qquad
     \bigl|2(\Th_t)_{x,y}-(\Th_t)_{x,y+1}-(\Th_t)_{x,y-1}\bigr| \le \frac{24}{\hat\ell(t)}
     \quad (x \ne y). \]
  Both are explicit constants, hence stronger than the paper's $\prec$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-decay, lem:prefactor}
  Writing $L = n+1+j$, the kernel differences telescope:
  $\mathrm{kern}(n{+}1)-\mathrm{kern}(n) = A(1-\rho)(\rho^j-\rho^n)$ and
  $\mathrm{kern}(n{+}1)+\mathrm{kern}(n{-}1)-2\,\mathrm{kern}(n) = A(1-\rho)^2(\rho^{n-1}+\rho^{j})$.
  Each lattice difference costs exactly one factor $1-\rho \asymp (1-t)^{1/2}$, i.e. one
  inverse decay length.  \emph{The diagonal $x = y$ is genuinely different}: there the
  kernel $\rho^d+\rho^{L-d}$ has a cusp and only one factor $1-\rho$ comes out, giving
  $4\|A(1-\rho)\| = O(1)$ --- consistent with the paper's right-hand side
  $1/(\|x-y\|+1)$, which is $1$ at $x = y$.

  \emph{A gap found by re-reading the paper's statement}: the bound $24/\hat\ell$ carries no
  decay, and therefore does \emph{not} imply the paper's $1/(\|x-y\|+1)$ once
  $\|x-y\|\gg\hat\ell$.  Keeping the factor $\rho^{\|x-y\|-1}$ closes it:
  $\bigl|2\Th_{x,y}-\Th_{x,y+1}-\Th_{x,y-1}\bigr| \le 60/(\|x-y\|+1)$, proved in the regime
  $\hat\ell = L$ from $\|x-y\|\le L/2$ and in the regime $\hat\ell = (1-t)^{-1/2}$ from
  $u e^{-u}\le e^{-1}\le 1/2$.
\end{proof}

\begin{definition}[Loops and cut-and-glue operators, Definitions 2.9--2.10]\label{def:loops}
  \lean{RBM.LoopIdx, RBM.LoopIdx.cutGlue, RBM.LoopIdx.cutGlueL, RBM.LoopIdx.cutGlueR}

  \begin{align*}
    \Gcal^{(a)}_k &: (\sigma_1,\dots,\sigma_k,\sigma_k,\dots,\sigma_n),\
      (a_1,\dots,a_{k-1},a,a_k,\dots,a_n),\\
    \Gcal^{(a),L}_{k,l} &: (\sigma_1,\dots,\sigma_k,\sigma_l,\dots,\sigma_n),\

\end{definition}

\begin{definition}[$G$-loops, Definition 2.9 and (2.41)]\label{def:gloop}
  \lean{RBM.Gsig, RBM.Gsig_conjTranspose, RBM.green_sub_green, RBM.green_sub_green_conj,
    RBM.trace_green_sub_trace_green_conj, RBM.gloopProd, RBM.gloop,
    RBM.gloopProd_append, RBM.gloop_rotate, RBM.sum_gloop_head, RBM.gloop_two,
    RBM.green_sub_green_conj', RBM.trace_green_sub_trace_green_conj',
    RBM.sum_gloop_two_ward, RBM.sum_block_ite, RBM.gloop_two_plus_minus,
    RBM.gloop_two_plus_minus_blocks, RBM.gloop_two_plus_minus_nonneg,
    RBM.im_green_apply_eq_mul_sum_normSq, RBM.normSq_green_le_im_green}
  \leanok\uses{def:model, def:loops, eq:2.10}
  For a fixed deterministic Hermitian $H$, $G(+) = (H-z)^{-1}$, $G(-) = (H-\bar z)^{-1}$ and
  \[ \mathcal L_{\sigma,\mathbf a} = \Bigl\langle \prod_i G(\sigma_i) E_{a_i} \Bigr\rangle ,
     \qquad \langle A\rangle = \operatorname{Tr} A . \]
  Proved here: $G(\sigma)^\dagger = G(-\sigma)$; the resolvent identity
  $G(z)-G(w) = (z-w)G(z)G(w)$, which is the source of every Ward identity downstream;
  and invariance of the loop under rotating its index data (trace cyclicity), which is
  the algebraic content of the word ``loop''.  Nothing here is probabilistic: the
  expectation and the It\^o calculus act on top of this layer.

  Also proved: the Ward identity $G(z)-G(\bar z) = 2i\eta\,G(z)G(\bar z)$ and its traced
  form $\operatorname{Tr}(G(z)A)-\operatorname{Tr}(G(\bar z)A) = 2i\eta\operatorname{Tr}(G(z)G(\bar z)A)$;
  and the one-index sum rule $\sum_b \mathcal L_{\sigma,(b,\mathbf a)} = W^{-1}\langle G(\sigma_1)\cdots\rangle$,
  which is $\sum_a E_a = W^{-1}I$ and is the first step of the paper's Ward identity for loops.
  Putting the two together gives the Ward identity for the $2$-loop,
  \[ 2i\eta\sum_b \mathcal L_{(+,-),(a,b)}
     = W^{-1}\bigl(\operatorname{Tr}(G(z)E_a)-\operatorname{Tr}(G(\bar z)E_a)\bigr) , \]
  a quadratic quantity on the left and a linear one on the right --- the identity that
  makes the local law self-improving.

  Finally, the $(+,-)$ $2$-loop is computed in closed form,
  \[ \mathcal L_{(+,-),(a,b)} = W^{-2}\sum_{\beta,\alpha}
     \bigl|G_{(b,\beta),(a,\alpha)}\bigr|^2 \ \ge 0 , \]
  which is the bridge to delocalization: the loop hierarchy evolves exactly the mass of
  the Green's function between two blocks.  Specialising the same identity to a single
  column gives the Ward identity in the form used everywhere,
  $\operatorname{Im}G_{qq} = \eta\sum_p |G_{pq}|^2$, hence
  $\eta|G_{pq}|^2 \le \operatorname{Im}G_{qq}$.
\end{definition}

\begin{lemma}[Cyclic invariance of $\Kcal$]\label{lem:K-cyclic}
  \lean{RBM.LoopIdx.rot, RBM.primRhs_rot, RBM.primInit_rot, RBM.rot_eq_on_level,

  $K_{t,(\sigma_2,\dots,\sigma_n,\sigma_1),(a_2,\dots,a_n,a_1)}=K_{t,\sigma,a}$.

\end{lemma}
\begin{proof}\leanok\uses{def:primitive, def:loops}

  and induction on $n$.
\end{proof}

\begin{lemma}[Eigenvector bound (2.10), proof of Theorem 2.2]\label{eq:2.10}
  \lean{RBM.green, RBM.green_eq_spectral, RBM.im_green_apply_self,
    RBM.sq_norm_eigenvector_le_sum, RBM.sum_eq_mul_im_green,
    RBM.sq_norm_eigenvector_le_im_green, RBM.sq_norm_eigenvector_le_of_norm_green_le}
  \leanok
  Let $H$ be Hermitian with eigenpairs $(\lambda_l,\psi_l)$ and $G(z)=(H-z)^{-1}$.
  For $\eta>0$,
  \[ |\psi_k(x)|^2 \le \sum_l \frac{\eta^2|\psi_l(x)|^2}{(\lambda_k-\lambda_l)^2+\eta^2}
     = \eta\,\operatorname{Im} G_{xx}(\lambda_k+i\eta). \]
  Hence $|G_{xx}(\lambda_k+i\eta)|\le C$ implies $|\psi_k(x)|^2\le C\eta$.
\end{lemma}
\begin{proof}\leanok
  $G = U\operatorname{diag}((\lambda-z)^{-1})U^*$ gives
  $\operatorname{Im}G_{xx}(E+i\eta)=\sum_l \eta|\psi_l(x)|^2/((\lambda_l-E)^2+\eta^2)$;
  at $E=\lambda_k$ the $l=k$ term alone is $|\psi_k(x)|^2/\eta$.
\end{proof}

\begin{remark}\label{rem:2.2}
  Theorem 2.2 itself follows by feeding in the local law (Theorem 2.3) at
  $\eta=N^{-1+\tau}$, which gives $|G_{xx}|=O(1)$ with high probability.  That input
  is probabilistic and belongs to the stochastic layer; the deterministic step above
  is what is formalized.
\end{remark}
