\chapter{The model}

\begin{definition}[Block covariance matrix, \S2.1]\label{def:SB}
  \lean{RBM.SB}\leanok
  Let $L\ge 3$ and work on $\ZL$.  Set
  \[ \SB_{ab} \;=\; \tfrac13\,\mathbf 1\!\left(\operatorname{dist}_{\ZL}(a,b)\le 1\right),
     \qquad a,b\in\ZL . \]
  Equivalently $\SB$ is the circulant matrix generated by the kernel supported on
  $\{0,1,-1\}$ with value $1/3$.
\end{definition}

\begin{remark}\label{rem:L3}
  The paper leaves the hypothesis $L\ge 3$ implicit.  It is needed: for $L=1$ the
  row sums are $1/3$, and for $L=2$ one has $1=-1$ in $\ZL$ so the row sums are
  $2/3$.  See \texttt{docs/paper-deltas.md}.
\end{remark}

\begin{lemma}[Elementary properties of $\SB$]\label{lem:SB-basic}
  \lean{RBM.SB_isSymm, RBM.SB_apply_add_right, RBM.sum_SB_row, RBM.SB_mulVec_one}
  \leanok\uses{def:SB}
  $\SB$ is symmetric and translation invariant, each row sums to $1$, and
  $\SB \mathbf 1 = \mathbf 1$.
\end{lemma}
\begin{proof}\leanok\uses{def:SB}
  Symmetry and translation invariance are immediate from the circulant form.  The
  row sums reduce, after the change of variables $b\mapsto a-b$, to the total mass
  of the kernel, which is $3\cdot\frac13=1$ because $0,1,-1$ are distinct in $\ZL$
  when $L\ge 3$.
\end{proof}

\begin{definition}[Deterministic domination, Definition 2.1 (ii)]\label{def:2.1-ii}
  \lean{RBM.UnifDetDom, RBM.DetDom, RBM.detDom_iff}\leanok
  For deterministic $N$-dependent quantities $f,g$ write $f\prec g$ if
  $f\le N^\tau g$ for every $\tau>0$ and all sufficiently large $N$.  The Lean
  definition is uniform in a possibly $N$-dependent parameter $u\in U(N)$, as in
  part (i) of the definition.
\end{definition}

\begin{lemma}[Closure properties of $\prec$]\label{lem:2.1-ii-closure}
  \lean{RBM.UnifDetDom.refl, RBM.UnifDetDom.trans, RBM.UnifDetDom.add,
    RBM.UnifDetDom.mul, RBM.UnifDetDom.const_mul_left, RBM.UnifDetDom.const_mul_right,
    RBM.UnifDetDom.of_eventually_le_const_mul}
  \leanok\uses{def:2.1-ii}
  On non-negative quantities $\prec$ is reflexive and transitive, closed under sums
  and products, and absorbs positive constant factors on either side; $f\le Cg$
  implies $f\prec g$.
\end{lemma}
\begin{proof}\leanok\uses{def:2.1-ii}
  Split $N^\tau = N^{\tau/2}N^{\tau/2}$ for transitivity and products; for
  constants use $N^\tau\to\infty$.
\end{proof}

\chapter{The propagator}

\begin{definition}[Propagator, Definition 2.13]\label{def:Theta}
  \lean{RBM.Theta}\leanok\uses{def:SB}
  For $\xi\in\CC$ with $|\xi|<1$ set $\Th_\xi := (1-\xi\SB)^{-1}$.
\end{definition}

\begin{lemma}[Operator norm of $\SB$]\label{lem:norm-SB}
  \lean{RBM.norm_SB}\leanok\uses{lem:SB-basic}
  $\|\SB\|_{\ell^\infty\to\ell^\infty}=1$.
\end{lemma}
\begin{proof}\leanok\uses{lem:SB-basic}
  The $\ell^\infty$ operator norm is the maximum $\ell^1$ row norm; all entries are
  non-negative and every row sums to $1$.
\end{proof}

\begin{lemma}[$\Th_\xi$ is a two-sided inverse]\label{lem:Theta-inv}
  \lean{RBM.Theta_mul, RBM.mul_Theta, RBM.eq_Theta_of_mul}\leanok
  \uses{def:Theta, lem:norm-SB}
  For $|\xi|<1$ the matrix $1-\xi\SB$ is invertible and $\Th_\xi$ is its two-sided
  inverse.  Moreover any left inverse of $1-\xi\SB$ equals $\Th_\xi$.
\end{lemma}
\begin{proof}\leanok\uses{lem:norm-SB}
  $\|\xi\SB\|=|\xi|<1$, so the geometric series converges and $1-\xi\SB$ is a unit.
\end{proof}

\begin{lemma}[Lemma 2.14 (1): symmetry]\label{lem:Theta-sym}
  \lean{RBM.Theta_transpose}\leanok\uses{lem:Theta-inv, lem:SB-basic}
  $(\Th_\xi)_{xy}=(\Th_\xi)_{yx}$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  $\Th^{\mathsf T}(1-\xi\SB) = \Th^{\mathsf T}(1-\xi\SB)^{\mathsf T}
   = \bigl((1-\xi\SB)\Th\bigr)^{\mathsf T} = 1$, so $\Th^{\mathsf T}$ is a left
  inverse; conclude by uniqueness.
\end{proof}

\begin{lemma}[Lemma 2.14 (2): translation invariance]\label{lem:Theta-transl}
  \lean{RBM.Theta_apply_add_right}\leanok\uses{lem:Theta-inv, lem:SB-basic}
  $(\Th_\xi)_{xy}=(\Th_\xi)_{x+1,y+1}$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Conjugating by the shift, i.e.\ taking the submatrix along $a\mapsto a+c$,
  fixes $1-\xi\SB$; conclude by uniqueness of the inverse.
\end{proof}

\begin{lemma}[Lemma 2.14 (3): commutativity]\label{lem:Theta-comm}
  \lean{RBM.Theta_commute_SB, RBM.Theta_commute}\leanok\uses{lem:Theta-inv}
  $[\SB,\Th_\xi]=[\Th_{\xi'},\Th_\xi]=0$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Inverses of commuting elements commute.
\end{proof}

\begin{lemma}[Lemma 2.14 (5): random walk representation]\label{lem:Theta-series}
  \lean{RBM.Theta_eq_tsum}\leanok\uses{lem:Theta-inv}
  $\Th_\xi=\sum_{k\ge 0}\xi^k(\SB)^k$, convergent in operator norm.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  Neumann series.
\end{proof}

\begin{lemma}[Row sums]\label{lem:Theta-rowsum}
  \lean{RBM.sum_Theta_row, RBM.Theta_mulVec_one}\leanok
  \uses{lem:Theta-inv, lem:SB-basic}
  $\sum_b (\Th_\xi)_{ab} = (1-\xi)^{-1}$.  Used in (3.47)--(3.48) and in Step 4 of
  the proof of Lemma 3.6.
\end{lemma}
\begin{proof}\leanok\uses{lem:SB-basic}
  Apply $\Th_\xi$ to the constant vector and use $\SB\mathbf 1=\mathbf 1$.
\end{proof}

\begin{lemma}[Equation (2.51)]\label{lem:Theta-deriv}
  \lean{RBM.hasDerivAt_Theta_apply, RBM.Theta_sub_Theta}\leanok
  \uses{lem:Theta-inv}
  $\partial_\xi \Th_\xi = \Th_\xi \SB \Th_\xi$, entrywise.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-inv}
  The resolvent identity
  $\Th_\zeta-\Th_\xi=(\zeta-\xi)\,\Th_\zeta\SB\Th_\xi$
  together with continuity of $\xi\mapsto\Th_\xi$.
\end{proof}

\begin{lemma}[$\ell^1$ and $\ell^\infty$ bounds, (3.35)--(3.36)]\label{lem:Theta-bounds}
  \lean{RBM.norm_Theta_le, RBM.sum_norm_Theta_row_le, RBM.norm_Theta_apply_le}\leanok
  \uses{lem:Theta-series, lem:norm-SB}
  $\max_a\sum_b |(\Th_\xi)_{ab}| \le (1-|\xi|)^{-1}$, and in particular
  $|(\Th_\xi)_{ab}| \le (1-|\xi|)^{-1}$.
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-series}
  Sum the Neumann series in operator norm.
\end{proof}

\begin{lemma}[Band structure and a first decay bound]\label{lem:Theta-band}
  \lean{RBM.SB_pow_apply_eq_zero, RBM.Theta_eq_geom_add, RBM.norm_Theta_apply_le_pow}\leanok
  \uses{lem:Theta-series, lem:Theta-bounds}
  $(\SB)^k$ vanishes outside the band of width $k$, and consequently
  \[ \bigl|(\Th_\xi)_{xy}\bigr| \;\le\;
     \frac{|\xi|^{\operatorname{dist}(x,y)}}{1-|\xi|} . \]
  This is exponential decay with length $1/\log(1/|\xi|)$, weaker than (2.52).
\end{lemma}
\begin{proof}\leanok\uses{lem:Theta-series}
  Truncate the Neumann series at order $d=\operatorname{dist}(x,y)$:
  $\Th_\xi=\sum_{k<d}(\xi\SB)^k+(\xi\SB)^d\Th_\xi$, and the finite sum has
  vanishing $(x,y)$ entry.
\end{proof}

\begin{lemma}[Fourier representation, Appendix B (B.1)]\label{eq:B.1}
  \lean{RBM.Shat, RBM.Shat_eq_cos, RBM.SB_mulVec_char, RBM.one_sub_mul_Shat_ne_zero,
    RBM.fourierKernel, RBM.Theta_eq_circulant_fourierKernel, RBM.Theta_apply_fourier}
  \leanok\uses{def:Theta, lem:Theta-inv}
  The plane wave $e_p(x)=e^{ipx}$, $p\in\frac{2\pi}{L}\ZL$, is an eigenvector of
  $\SB$ with eigenvalue $\widehat S(p)=(1+2\cos p)/3$.  Since $|\widehat S(p)|\le 1$,
  $1-\xi\widehat S(p)\neq 0$ for $|\xi|<1$, and
  \[ (\Th_\xi)_{xy} \;=\; \frac1L\sum_{p\in\frac{2\pi}{L}\ZL}
     \frac{e^{ip(x-y)}}{1-\xi\widehat S(p)} . \]
\end{lemma}
\begin{proof}\leanok\uses{def:Theta, lem:Theta-inv}
  Call the right-hand side $K(x-y)$.  By the eigenvector relation and orthogonality
  of characters, $K-\xi\,\SB K=\delta_0$; the circulant matrix generated by $K$ is
  therefore a left inverse of $1-\xi\SB$, and uniqueness of the inverse closes it.
\end{proof}

\begin{lemma}[The characteristic root $\rho(\xi)$]\label{lem:rho}
  \lean{RBM.rho, RBM.norm_root_ne_one, RBM.norm_rho_lt_one}\leanok\uses{def:Theta}
  Write $c = 3/\xi - 1$.  The characteristic equation of the three-term recursion
  behind $(1-\xi\SB)\Th_\xi = 1$ is $\rho^2 - c\rho + 1 = 0$, whose two roots
  multiply to $1$.  For $0 < |\xi| < 1$ neither root lies on the unit circle, so
  exactly one has modulus $< 1$; call it $\rho(\xi)$.
\end{lemma}
\begin{proof}\leanok\uses{def:Theta}
  If $|\rho| = 1$ then $|c| = |\rho + \rho^{-1}| \le |\rho| + |\rho^{-1}| = 2$,
  hence $3/|\xi| = |c+1| \le 3$, i.e.\ $|\xi| \ge 1$.  This is the same statement
  as $1 - \xi\widehat S(p) \ne 0$ in the Fourier picture of Appendix B, and it is
  exactly what fails for $|\xi| \ge 1$.
\end{proof}

\begin{lemma}[Closed form of the propagator]\label{lem:closed-form}
  \lean{RBM.theta_eq_circulant, RBM.theta_apply_closed_form}\leanok
  \uses{lem:rho, lem:Theta-inv}
  For $0 < |\xi| < 1$,
  \[ (\Th_\xi)_{xy} = A(\xi)\bigl(\rho^{\,d} + \rho^{\,L-d}\bigr),\qquad
     d = \overline{(x-y)} \in [0,L),\qquad
     A(\xi) = \frac{3\rho}{\xi\,(\rho^L-1)(\rho^2-1)} . \]
\end{lemma}
\begin{proof}\leanok\uses{lem:rho, lem:Theta-inv}
  The kernel satisfies $k(u-1) + k(u+1) - c\,k(u) = -(3/\xi)\delta_{u,0}$.  The
  symmetric shape makes the value at $n=L$ agree with the value at $n=0$, so the
  wrap-around at $d = L-1$ falls under the generic case and $d = 0$ is the only
  special one; the defect there is what fixes $A(\xi)$.  Conclude by uniqueness
  of the inverse.
\end{proof}

\begin{lemma}[Decay rate for real spectral parameter]\label{lem:rho-rate}
  \lean{RBM.rho_real_bounds}\leanok\uses{lem:rho}
  For $0 < t < 1$, $\rho(t)$ is real, lies in $(0,1)$, and
  \[ \sqrt{1-t} \;\le\; 1 - \rho(t) \;\le\; \sqrt3\,\sqrt{1-t} . \]
  So the decay length $1/(1-\rho)$ is of order $(1-t)^{-1/2}$.  This is the
  long-edge case $\xi = t|m|^2$, where $|m| = 1$ makes $\xi$ real; it is the case
  that needs the sharp length.
\end{lemma}
\begin{proof}\leanok\uses{lem:rho}
  The discriminant $(3/t-1)^2 - 4$ is a positive real, so both roots are real and
  $\rho \in (0,1)$; then $1 - \|\rho\| = 1 - \rho$, which is the direction the
  decay estimate needs.  Combine with the exact identity
  $(1-\rho)^2 = (1-\xi)(1+\rho+\rho^2)$ and $1 \le 1+\rho+\rho^2 \le 3$.
\end{proof}

\begin{lemma}[Lemma 2.14 (4): exponential decay, (2.52)]\label{lem:Theta-decay}
  \uses{lem:Theta-band, lem:closed-form, lem:rho-rate}
  With $\hat\ell(\xi):=\min(|1-\xi|^{-1/2},L)$,
  \[ \bigl|(\Th_\xi)_{xy}\bigr| \;\le\;
     \frac{C\,e^{-c|x-y|/\hat\ell(\xi)}}{|1-\xi|\cdot\hat\ell(\xi)} . \]
  \emph{In progress.}  The root $\rho(\xi)$ is constructed
  (Lemma~\ref{lem:rho}); what remains is the closed form
  \[ (\Th_\xi)_{xy} \;=\; A(\xi)\bigl(\rho^{\,d}+\rho^{\,L-d}\bigr),
     \qquad d = \overline{(x-y)} \in [0,L), \qquad
     A(\xi) = \frac{3\rho}{\xi\,(\rho^L-1)(\rho^2-1)} , \]
  verified by substitution into $(1-\xi\SB)\Th_\xi=1$ and closed by uniqueness of
  the inverse (Lemma~\ref{lem:Theta-inv}).  Summing it over $d$ gives
  $3\rho/[\xi(\rho-1)^2]$, which the characteristic equation turns into exactly
  $(1-\xi)^{-1}$, matching the independently proved Lemma~\ref{lem:Theta-rowsum};
  this cross-check confirms the constant.  Setting $\xi = 1-\varepsilon$ and
  $\rho = 1-\delta$ gives $\delta \approx \sqrt{3}\,|1-\xi|^{1/2}$, which is the
  paper's decay length.  This route avoids the contour shift and Poisson
  summation of Appendix B entirely.
\end{lemma}

\begin{remark}\label{rem:stochastic}
  Sections 2.4, 2.6, 2.7 and 5--7 of the paper rest on It\^o calculus for matrix
  Brownian motion, which Mathlib does not have: there is no It\^o formula, no
  matrix-valued SDE theory, and no Dyson Brownian motion.  The loop hierarchy
  (2.45), Lemmas 2.18--2.20 and Theorem 2.21 are therefore left as stated
  hypotheses, so that the logical skeleton of the main theorems still typechecks.
\end{remark}
